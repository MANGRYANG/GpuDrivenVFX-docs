```C
void App::Render()
{
    // 프레임 드로우 준비 및 배경 색 초기화 (#0D141F)
    m_renderer.BeginFrame(0.05f, 0.08f, 0.12f, 1.0f);
	
    // 셰이더 바인딩
    m_shader.Bind(m_renderer.GetContext());

```

**Render 단계**에서 렌더링하던 기존 삼각형 메쉬(`m_triangleMesh`) 대신, 
사각형 메쉬(`m_quadMesh`)를 렌더링하도록 코드를 교체하였습니다.

**mesh.h**

```C
// Mesh를 구성하는 정점 리소스 초기화 함수
bool Initialize
(
	ID3D11Device* device,
	ID3DBlob* vertexShaderBlob,
	const std::vector<Vertex>& vertices,
	const std::vector<uint32_t>& indices
);
```

`Initialize` 함수의 파라미터로 **인덱스 데이터를 저장하는 가변 배열의 주소**를 전달하도록 수정하였습니다.

```C
// 그려야 할 인덱스의 총 개수
UINT m_indexCount = 0;
```

기존에 존재하던 그려야 할 정점의 개수를 저장하는 **멤버 변수를 제거**하고(`m_vertexCount`),
그려야 할 인덱스의 총 개수를 저장하는 **멤버 변수를 추가**하였습니다.(`m_indexCount`)


**mesh.cpp**

```C
bool Mesh::Initialize
(
    ID3D11Device* device,
    ID3DBlob* vertexShaderBlob,
    const std::vector<Vertex>& vertices,
    const std::vector<uint32_t>& indices
)
{
    ...
	
    // 인덱스 버퍼 설명자 구조체
    D3D11_BUFFER_DESC indexBufferDesc = {};
    // 버퍼를 구성할 데이터 배열의 메모리 크기 설정
    indexBufferDesc.ByteWidth = 
	    static_cast<UINT>(sizeof(uint32_t) * indices.size());
    // 버퍼의 사용 방식을 생성 후 변경 불가능한 읽기 전용으로 설정
    indexBufferDesc.Usage = D3D11_USAGE_IMMUTABLE;
    // 버퍼의 사용 용도를 파이프라인의 인덱스 버퍼로 지정
    indexBufferDesc.BindFlags = D3D11_BIND_INDEX_BUFFER;
    // CPU가 버퍼에 직접 엑세스할 수 없도록 설정
    indexBufferDesc.CPUAccessFlags = 0;
    // 기타 특수 기능 사용하지 않음
    indexBufferDesc.MiscFlags = 0;
    // Structured Buffer가 아니므로 기본값으로 설정
    indexBufferDesc.StructureByteStride = 0;
	
    initialData = {};
    // 인덱스 데이터 배열을 초기화 데이터로 설정
    initialData.pSysMem = indices.data();
	
    // 인덱스 버퍼 생성
    hr = device->CreateBuffer
    (
        &indexBufferDesc,
        &initialData,
        m_indexBuffer.GetAddressOf()
    );
	
    // 버퍼 생성에 실패한 경우
    if (FAILED(hr))
    {
        // 메시지 박스 플로팅
        MessageBoxW
        (
	        nullptr,
	        L"Failed to create index buffer.",
	        L"Mesh Error",
	        MB_OK | MB_ICONERROR
	    );
		
        return false;
    }
    ...
	
    // 정점 바이트 크기 멤버 변수 초기화
    m_vertexStride = sizeof(Vertex);
    // 인덱스 개수 멤버 변수 초기화
    m_indexCount = static_cast<UINT>(indices.size());
	
    return true;
}
```

`Initialize` 함수에서 다음 작업을 수행할 수 있도록 코드를 추가하였습니다.
- **인덱스 버퍼 설명자 구조체**(`D3D11_BUFFER_DESC`) 생성
- **인덱스 버퍼 생성**
- **인덱스 개수**를 담는 멤버 변수(`m_indexCount`) 초기화


```C
void Mesh::Draw(ID3D11DeviceContext* context)
{
    ...
	
    // 입력 조립기 단계의 인덱스 버퍼 설정
    context->IASetIndexBuffer
    (
        m_indexBuffer.Get(),  // 인덱스 버퍼
        DXGI_FORMAT_R32_UINT, // 인덱스 버퍼의 데이터 형식
        0                     // 버퍼 시작점으로부터의 바이트 오프셋
    );
	
    ...
    
    // Draw call 호출
    context->DrawIndexed
    (
        m_indexCount,   // 그려야 할 인덱스의 총 개수
        0,              // 버퍼에서 처음으로 읽을 인덱스
        0               // 정점 버퍼에 더할 기준 정점 오프셋
    );
}
```

`Draw` 함수에서, **입력 조립기(Input Assembler, IA) 단계에 인덱스 버퍼를 바인딩**하는 코드를 추가하였습니다.

기존에 사용하던 디바이스 컨텍스트의 `Draw` 함수를 `DrawIndexed` 함수로 교체하여, **인덱스 버퍼를 기반으로 삼각형들을 조립하여 사각형을 그리도록 수정**하였습니다.

실행 결과, 2개의 삼각형을 합쳐 사각형을 그리는 것을 확인하였습니다.
이를 통해 **추후 구현할 파티클 시스템의 기본 단위로 활용될 사각형 메쉬와 데이터 구조가 올바르게 해석됨을 검증**하였습니다.
