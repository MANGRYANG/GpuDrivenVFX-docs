# 주요 내용

## 💻 코드

**mesh.h**

```C
#pragma once

class Mesh
{
public:
    // Mesh 클래스 생성자
    Mesh() = default;
    // Mesh 클래스 소멸자
    ~Mesh() = default;
	
    // DirectX 11 최소 렌더링 파이프라인 검증용 삼각형 정점 리소스 초기화 함수
    bool InitializeTriangle(ID3D11Device* device, ID3DBlob* vertexShaderBlob);
	
    // Mesh를 그리기 위한 그래픽 파이프라인 설정 및 Draw call 호출 함수
    void Draw(ID3D11DeviceContext* context);

private:
    // GPU 메모리에 생성되는 정점 데이터 버퍼
    Microsoft::WRL::ComPtr<ID3D11Buffer> m_vertexBuffer;
    // 정점 버퍼의 데이터 구조를 정점 셰이더에 매칭하는 입력 레이아웃 객체
    Microsoft::WRL::ComPtr<ID3D11InputLayout> m_inputLayout;
	
    // 정점 한 개가 차지하는 바이트 크기
    UINT m_vertexStride = 0;
    // 버퍼에 저장된 총 정점의 개수
    UINT m_vertexCount = 0;
};
```

**Mesh 클래스**를 정의하였습니다.

```C
bool Mesh::InitializeTriangle(ID3D11Device* device, ID3DBlob* vertexShaderBlob)
{
    ...
	
    // 삼각형을 구성할 정점 데이터를 담은 배열
    const Vertex vertices[] =
    {
        {
            DirectX::XMFLOAT3(0.0f, 0.5f, 0.0f),

        }
    };
	
    // 버퍼 설명자 구조체
    D3D11_BUFFER_DESC vertexBufferDesc = {};
    ...
	
    // 정점 버퍼 생성
    HRESULT hr = device->CreateBuffer
    (
        &vertexBufferDesc,
        &initialData,
        m_vertexBuffer.GetAddressOf()
    );
	
    ...
	
    // 입력 레이아웃 설명자 구조체 배열
    const D3D11_INPUT_ELEMENT_DESC inputLayoutDesc[] =
    {
        {
            "POSITION",                     // 의미 체계명
            0,                              // 의미 체계 인덱스
            DXGI_FORMAT_R32G32B32_FLOAT,    // 데이터 포맷
            0,                              // 파이프라인 입력 슬롯 번호
            0,                              // 바이트 오프셋
            D3D11_INPUT_PER_VERTEX_DATA,    // 입력 데이터 클래스 식별
            0                               // 정점별 데이터인 경우 0으로 설정
        },
        {
            "NORMAL",                       // 의미 체계명
            0,                              // 의미 체계 인덱스
            DXGI_FORMAT_R32G32B32_FLOAT,    // 데이터 포맷
            0,                              // 파이프라인 입력 슬롯 번호
            D3D11_APPEND_ALIGNED_ELEMENT,   // 바이트 오프셋 자동 정렬 배치
            D3D11_INPUT_PER_VERTEX_DATA,    // 입력 데이터 클래스 식별
            0                               // 정점별 데이터인 경우 0으로 설정
        },
        {
            "COLOR",                        // 의미 체계명

        }

    // 입력 레이아웃 객체 생성
    hr = device->CreateInputLayout
    (

        inputLayoutDesc,

        // 정점 셰이더 바이트코드의 크기
        vertexShaderBlob->GetBufferSize(),

        m_inputLayout.GetAddressOf()
    );
	
    ...
	
    // 정점 바이트 크기 멤버 변수 초기화
    m_vertexStride = sizeof(Vertex);
    // 정점 개수 멤버 변수 초기화
    m_vertexCount = ARRAYSIZE(vertices);
	
    return true;
}
```

 삼각형의 **세 정점에 대한 정점 데이터를 저장하는 배열**을 생성하였습니다. (`vertices`)

이후 해당 데이터를 GPU에 전달하기 위해 디바이스를 통해 **정점 버퍼를 생성**하고, 데이터 구조를 정점 셰이더 규격에 맞춰 올바르게 해석할 수 있도록 **입력 레이아웃 객체를 생성**하도록 구현하였습니다.

```C
void Mesh::Draw(ID3D11DeviceContext* context)
{
    ...
	
    // 시작 오프셋 (0으로 설정)
    UINT offset = 0;
	
    // 파이프라인 입력 슬롯에 전달할 정점 버퍼 포인터 배열
    ID3D11Buffer* vertexBuffers[] =
    {
        // 0번 슬롯에 할당할 정점 버퍼
        m_vertexBuffer.Get()
    };
	
    // 입력 조립기 단계의 입력 레이아웃 설정
    context->IASetInputLayout(m_inputLayout.Get());
	
    // 입력 조립기 단계의 정점 버퍼 설정
    context->IASetVertexBuffers
    (
        0,                  // 버퍼를 바인딩할 입력 슬롯 번호
        1,                  // 설정할 버퍼의 개수
        vertexBuffers,      // 정점 버퍼 포인터들이 담긴 배열
        &m_vertexStride,    // 정점 한 개가 차지하는 바이트 크기
        &offset             // 버퍼 시작점으로부터의 바이트 오프셋
    );
	
    // 입력 조립기 단계의 데이터 형식 설정 
    context->IASetPrimitiveTopology
    (
        // 정점 데이터를 삼각형 리스트로 해석
        D3D11_PRIMITIVE_TOPOLOGY_TRIANGLELIST
    );
	
    // Draw call 호출
    context->Draw
    (
        m_vertexCount,  // 그릴 정점의 총 개수
        0               // 버퍼에서 처음으로 읽을 정점의 인덱스
    );
}
```

**입력 조립기(Input Assembler, IA)** 단계에 **입력 레이아웃과 정점 버퍼를 설정**하고 **데이터 형식을 지정**하여 GPU가 정점 데이터를 해석할 수 있도록 준비하는 로직을 구현하였습니다.

이후 **디바이스 컨텍스트의 `Draw` 함수를 호출**해 GPU에 `Draw call` 명령을 전달함으로써 삼각형을 그릴 수 있도록 작성하였습니다.


**app.cpp**

```C
void App::Render()
{
    // 프레임 드로우 준비 및 배경 색 초기화 (#0D141F)
    m_renderer.BeginFrame(0.05f, 0.08f, 0.12f, 1.0f);
	
    // 셰이더 바인딩
    m_shader.Bind(m_renderer.GetContext());
	
    // 검증용 삼각형 그리기
    m_triangleMesh.Draw(m_renderer.GetContext());
	
    // 최종 화면 출력
    m_renderer.EndFrame();
}
```

**애플리케이션의 Render 단계**에서 정점 및 픽셀 셰이더를 바인딩하여 그래픽 파이프라인의 처리 방식을 결정한 직후, `Mesh::Draw` 함수를 호출하여 삼각형이 화면에 그려지도록 구현하였습니다.

실행 결과, 의도한 대로 삼각형이 그려지는 것을 확인하였습니다.
이를 통해 **DirectX 11의 최소 렌더링 파이프라인이 정상적으로 구축되었음이 검증**되었습니다.
